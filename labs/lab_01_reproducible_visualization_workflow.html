# Part 7: Create a Simulated Public Health Dataset

```{r}
#| label: create-simulated-health-data
#| echo: false

# Set seed (for reproducibility, including the manual data)

set.seed(716)

# 12 fictional counties, regions, and raw values defined manually

county_health_data <- tibble(
  county = c(
    "Asael County", "Iva County", "Beauty County", "Lantern County",  # North
    "Dane County", "Clover County", "Ziven County", "Cross County",   # Central
    "Fur County", "Famous County", "Han County", "Cookieville County"    # South
  ),
  region = c(
    "North", "North", "North", "North",
    "Central", "Central", "Central", "Central",
    "South", "South",
    "South", "South"
  ),
  adult_population = c(
    40000, 120000, 85000, 32000,  # North popu.
    210000, 65000, 95000, 150000,   # Central popu.
    55000, 78000, 110000, 240000   # South popu. 
  ),
  asthma_visits = c(
    310, 820, 540, 210,  # North visits
    1450, 480, 610, 1100,  # Central visits
    390, 520, 790, 1350    # South visits
  ),
  vaccination_rate = c(
    0.824, 0.761, 0.895, 0.642,   # North rates
    0.881, 0.715, 0.793, 0.854,   # Central rates
    0.612, 0.748, 0.805, 0.913   # South rates
  )
) %>%
  mutate(
    # in order to calculate the asthma visit rate per 100,000
    asthma_visit_rate = round((asthma_visits / adult_population) * 100000, 1)
  )

# Completed dataset displayed 

county_health_data

```

# Part 11: Create a Ranked Vaccination Bar Chart

```{r}
#| label: create-vaccination-plot
#| fig-cap: "Stimulated vaccination coverage by county."
#| fig-alt: "Horizontal bar chart ranking fictional counties by simulated vaccination coverage."
#| fig-width: 9
#| fig-height: 7
#| echo: false

# This reorder counties by vaccination rate and create a horizontal bar chart

vaccination_plot <- county_health_data %>%
  mutate(
    county = fct_reorder(county, vaccination_rate)
  ) %>%
  ggplot(
    aes(
      x = county, 
      y = vaccination_rate
    )
  ) + 
  geom_col(fill = "#009E73") +   # scientific standard for a colorblind-safe color (Wong), Bluish Green (from the Okabe-Ito Palette)
  geom_hline(
    yintercept = 0.80,
    linetype = "dashed",
    color = "gray40"
  ) + 
  coord_flip() +
  scale_y_continuous(labels = scales::percent_format(accuracy = 1)) +
  labs(
    title = "Simulated Vaccination Coverage by County",
    subtitle = "Horizontal ranking with 80% target reference line",
    x = "County",
    y = "Vaccination Coverage",
    caption = "Data are simulated not for real-life use, only instructional use or teaching."
  ) +
  theme_minimal(base_size = 12)

# This prints the plot so it renders in the Quatro document

vaccination_plot
```

# Part 14: Independent Visualization

```{r}
#| label: independent-visualization
#| fig-cap: "Independent visualization analyzing adult population and total asthma visits created by the student."
#| fig-alt: "Scatterplot showing a positive relationship between adult population size and total emergency asthma visits across regions."
#| fig-width: 10
#| fig-height: 8
#| echo: false

# This creates a scatterplot that explores population vs. asthma burden

independent_plot <- county_health_data %>%
  ggplot(
    aes(
      x = adult_population,
      y = asthma_visits
    )) +    
  # Points that are colored by region using a colorblind-friendly design 
  geom_point(
    aes(
      color = region
    ), 
    size = 3.5, 
    alpha = 0.8
  ) +    
  # This is a dashed trend line which shows direction with no overbearing layout
  geom_smooth(
    method = "lm",
    se = FALSE,
    color = "midnightblue",
    linetype = "twodash",
    size = 1
  ) + 
  # This is a built-in colorblind palette from scale_color_brewer (Set2 is appropriate)
  scale_color_brewer(palette = "Set2") +
  # This adds comma formatting to large pop. numbers on the x-axis (readable with this)
  scale_x_continuous(labels = scales::comma_format()) + 
  labs(
    title = "Adult Population Size vs. Total Emergency Asthma Visits",
    subtitle = "Exploratory relationship analysis tracking volume across regions",
    x = "Adult Population Size (Count)",
    y = "Total Emergency Asthma Visits (Count)",
    color = "Geographic Region",
    caption = "Data are simulated not for real-life use."
  ) +
  theme_minimal(base_size = 12) +
  theme(
    legend.position = "bottom",
    plot.title = element_text(face = "bold")
  )

# This prints the plot

independent_plot 
```
